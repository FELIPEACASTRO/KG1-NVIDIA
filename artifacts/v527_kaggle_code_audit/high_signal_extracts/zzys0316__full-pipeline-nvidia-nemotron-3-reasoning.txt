def _pure_rmsnorm_fn(x, weight, bias=None, z=None, eps=1e-5,
                     group_size=None, norm_before_gate=True, upcast=True):
    dtype = x.dtype
    if upcast:
        x = x.float()
    variance = x.pow(2).mean(-1, keepdim=True)
    x_normed = x * torch.rsqrt(variance + eps)
    out = x_normed * weight.float()
    if bias is not None:
        out = out + bias.float()
    if z is not None:
        out = out * F.silu(z.float())
    return out.to(dtype)

def classify_equation_vs_symbol(prompt: str) -> str:
    q = prompt.strip()

    # extract the target expression after "determine the result for" if present
    m = re.search(r"determine the result for[:\s]*([^\n\r]*)", q, re.IGNORECASE)
    expr = m.group(1).strip() if m else q

    # counts
    digit_count = len(re.findall(r"\d", expr))
    alpha_count = len(re.findall(r"[A-Za-z]", expr))
    symbol_count = len(re.findall(r"[^\w\s]", expr))  # punctuation / symbols

    # rule 1: numeric_equation if expression contains two numeric groups separated by a non-digit operator
    if re.search(r"\d+\s*[^0-9\s]\s*\d+", expr):
        return "numeric_equation"

    # rule 2: symbol_transform if symbols dominate and digits/letters are scarce
    total_chars = max(1, len(expr))
    if symbol_count / total_chars > 0.5 and digit_count + alpha_count < max(2, symbol_count//2):
        return "symbol_transform"

    # fallback: look at examples in the prompt body (presence of lines with digits → numeric)
    if re.search(r"^\s*\d+[^=\n]*=", q, re.MULTILINE):
        return "numeric_equation"
    if re.search(r"[^\w\s]{2,}", q):  # repeated punctuation sequences
        return "symbol_transform"

    return "unknown"

def classify_puzzle(prompt: str) -> str:
    p = prompt.lower()

        return "gravity_physics"
    if "transformation rules" in p:
        return classify_equation_vs_symbol(prompt) if 'classify_equation_vs_symbol' in globals() else "symbol_transform"
    return "unknown"

import re
from typing import Optional, Tuple

def int_to_roman(num: int) -> str:
    """Convert integer to Roman numeral."""
    vals = [1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1]
    syms = ["M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"]
    out = ""
    for v, s in zip(vals, syms):
        while num >= v:
            out += s
            num -= v
    return out


def solve_numeral_with_trace(prompt: str) -> Tuple[Optional[str], Optional[str]]:

        return None, None
    
    num = int(m.group(1))

    return final_answer, trace

import re
import statistics
from typing import Optional, Tuple


def long_multiplication_lines(a_str: str, b_str: str) -> Tuple[list[str], str]:
    """Generate long multiplication step-by-step trace."""
    a = float(a_str)
    b = float(b_str)
    result = a * b
    
    lines = []
    lines.append(f"    {a_str}")
    lines.append(f"  × {b_str}")
    lines.append(f"  " + "-" * max(len(a_str), len(b_str) + 2))
    
    # For simplicity, show the result (full implementation would show partial products)
    result_str = f"{result:.4f}".rstrip('0').rstrip('.')
    lines.append(f"  {result_str}")
    
    return lines, result_str


def long_division_lines(dividend_str: str, divisor_str: str) -> Tuple[list[str], str]:
    """Generate long division step-by-step trace."""
    dividend = float(dividend_str)
    divisor = float(divisor_str)
    
    if divisor == 0:
        return [f"  {dividend_str} / {divisor_str} = undefined"], "undefined"
    
    result = dividend / divisor
    result_str = f"{result:.4f}".rstrip('0').rstrip('.')
    
    lines = []
    lines.append(f"  {dividend_str} ÷ {divisor_str}")
    lines.append(f"  = {result_str}")
    
    return lines, result_str


def truncate_3dp(num_str: str) -> str:
    """Truncate to 3 decimal places."""
    if '.' not in num_str:
        return num_str
    parts = num_str.split('.')
    if len(parts[1]) <= 3:
        return num_str
    return f"{parts[0]}.{parts[1][:3]}"


def solve_gravity_with_trace(prompt: str) -> Tuple[Optional[str], Optional[str]]:
    """
    Generate COT trace for gravity problem with explicit arithmetic.
    
    Args:
        prompt: The input prompt containing examples and query
        answer: The expected answer (ground truth)
    
    Returns:
        Tuple of (final_answer, trace)
    """
    # Extract examples: t = X.XXs, distance = Y.YY m
    pairs = re.findall(r"t = ([\d.]+)s, distance = ([\d.]+) m", prompt)
    if not pairs:
        return None, None
    
    lines: list[str] = []
    
    # Header
    lines.append("We need to determine the falling distance using d = k*t^2.")
    lines.append("Let me find k from the examples with explicit arithmetic.")
    lines.append("I will put my final answer inside \\boxed{}.")
    lines.append("")
    
    # Step 1: Calculate k from each example with EXPLICIT arithmetic
    lines.append("Step 1: Calculate gravitational constant k from each example")
    lines.append("Formula: k = d/t^2")
    lines.append("")
    
    k_strs: list[str] = []
    k_values: list[float] = []
    
    for i, (t_str, d_str) in enumerate(pairs, 1):
        t = float(t_str)
        d = float(d_str)
        
        if t <= 0:
            continue
        
        lines.append(f"Example {i}: t = {t_str}s, d = {d_str}m")
        
        # Calculate t^2 with explicit multiplication
        lines.append(f"  First, calculate t^2 = {t_str} × {t_str}:")
        mult_lines, t_sq_str = long_multiplication_lines(t_str, t_str)
        lines.extend(["  " + line for line in mult_lines])
        lines.append(f"  t^2 = {t_sq_str}")
        lines.append("")
        
        # Calculate k = d/t^2 with explicit division
        t_sq_truncated = truncate_3dp(t_sq_str)
        d_truncated = truncate_3dp(d_str)
        
        lines.append(f"  Then, calculate k = d/t^2 = {d_truncated} ÷ {t_sq_truncated}:")
        div_lines, k_str = long_division_lines(d_truncated, t_sq_truncated)
        lines.extend(["  " + line for line in div_lines])
        lines.append(f"  k = {k_str}")
        lines.append("")
        
        k_strs.append(k_str)
        k_values.append(float(k_str))
    
    if not k_strs:
        return None, None
    
    # Step 2: Find median k (more robust than mean)
    lines.append("Step 2: Determine representative k value")

import re
import statistics

def truncate_3dp(num_str: str) -> str:
    """Truncate to 3 decimal places."""
    if '.' not in num_str:
        return num_str
    parts = num_str.split('.')
    if len(parts[1]) <= 3:
        return num_str
    return f"{parts[0]}.{parts[1][:3]}"


def cast_dp_pair(a_str: str, b_str: str) -> Tuple[str, str, int, int]:
    """
    Cast two decimal numbers to same precision for division.
    Returns: (a_cast, b_cast, a_decimal_places, b_decimal_places)
    """
    a_dp = len(a_str.split('.')[-1]) if '.' in a_str else 0
    b_dp = len(b_str.split('.')[-1]) if '.' in b_str else 0
    
    # Convert to integers by removing decimal point
    a_int = a_str.replace('.', '')
    b_int = b_str.replace('.', '')
    
    return a_int, b_int, a_dp, b_dp


def long_division_lines(dividend_str: str, divisor_str: str) -> Tuple[List[str], str]:
    """Generate long division step-by-step trace."""
    dividend = float(dividend_str)
    divisor = float(divisor_str)
    
    if divisor == 0:
        return [f"  {dividend_str} ÷ {divisor_str} = undefined"], "undefined"
    
    result = dividend / divisor
    result_str = f"{result:.6f}".rstrip('0').rstrip('.')
    
    lines = []
    lines.append(f"  {dividend_str} ÷ {divisor_str}")
    lines.append(f"  = {result_str}")
    
    return lines, result_str


def long_multiplication_lines(a_str: str, b_str: str) -> Tuple[List[str], str]:
    """Generate long multiplication step-by-step trace."""
    a = float(a_str)
    b = float(b_str)
    result = a * b
    
    lines = []
    lines.append(f"    {a_str}")
    lines.append(f"  × {b_str}")
    lines.append(f"  " + "-" * max(len(a_str), len(b_str) + 2))
    
    result_str = f"{result:.6f}".rstrip('0').rstrip('.')
    lines.append(f"  {result_str}")
    
    return lines, result_str

def _word_pattern(word: str) -> Tuple[int, ...]:
    """
    Convert word to pattern based on character repetition.
    Example: 'hello' → (0,1,2,2,3), 'world' → (0,1,2,3,4)
    """
    seen: dict[str, int] = {}
    pattern: List[int] = []
    for char in word:
        if char not in seen:
            seen[char] = len(seen)
        pattern.append(seen[char])
    return tuple(pattern)


def _candidate_words_for_partial(
    partial: str,
    cipher_to_plain: dict[str, str],
    plain_to_cipher: dict[str, str],
    cipher_word: str,
) -> List[str]:
    """
    Find dictionary words matching a partial decryption.
    
    Args:
        partial: Partially decoded word with '?' for unknowns
        cipher_to_plain: Current cipher→plain mapping
        plain_to_cipher: Current plain→cipher mapping
        cipher_word: Original cipher text word
    
    Returns:
        List of candidate words from dictionary
    """
    candidates: List[str] = []
    target_len = len(partial)
    target_pattern = _word_pattern(cipher_word)
    
    for word in _load_wonderland():
        # Length check
        if len(word) != target_len:
            continue
        
        # Pattern check
        if _word_pattern(word) != target_pattern:
            continue
        
        # Partial match check
        match = True
        for i, ch in enumerate(partial):
            if ch != "?" and ch != word[i]:
                match = False
                break
        if not match:
            continue
        
        # Forward consistency check (cipher→plain)
        consistent = True
        for cc, wc in zip(cipher_word, word):
            if cc in cipher_to_plain and cipher_to_plain[cc] != wc:
                consistent = False
                break
        
        if consistent:
            candidates.append(word)
    
    candidates.sort()
    return candidates

import re
from typing import Optional, Tuple, List
from dataclasses import dataclass


@dataclass
class _Example:
    """Parsed example structure."""
    left: Tuple[str, str]    # Two-character left operand
    op: str                   # Single-character operator
    right: Tuple[str, str]    # Two-character right operand
    output: str               # Result string


def _box(s: str) -> str:
    """Wrap each character in 【】 brackets."""
    return "".join(f"【{c}】" for c in s)


def _quote(s: str) -> str:
    """Wrap string in 【】 brackets."""
    return f"【{s}】"


def _concat_type(examples: List[_Example]) -> Optional[str]:
    """
    Determine concatenation type from examples.
    
    Returns:
        'fwd' if all examples match forward concatenation (AB+CD = ABCD)
        'rev' if all examples match reverse concatenation (AB+CD = CDAB)
        None if neither pattern matches
    """
    if all(ex.output == ex.left[0] + ex.left[1] + ex.right[0] + ex.right[1] 
           for ex in examples):
        return "fwd"
    
    if all(ex.output == ex.right[0] + ex.right[1] + ex.left[0] + ex.left[1] 
           for ex in examples):
        return "rev"
    
    return None


def generate_symbol_transform_trace(prompt: str) -> Tuple[Optional[str], Optional[str]]:
    """
    Generate COT trace for symbol transformation (cryptarithm) puzzles.
    
    Focuses on concatenation operations (forward and reverse).
    Uses actual pattern matching rather than digit assignment.
    
    Args:
        prompt: Input containing examples and question
    
    Returns:
        Tuple of (final_answer, trace)
    """
    # Extract examples: "AB+CD = EFGH"
    example_lines = re.findall(r"([^\n=]+?)\s*=\s*([^\n]+)", prompt)
    if not example_lines:
        return None, None
    
    # Extract query: "determine the result for: AB+CD"
    query_match = re.search(r"determine the result for:\s*([^\n]+)", prompt, re.IGNORECASE)
    if not query_match:
        # Fallback: last line without '='
        lines = [l.strip() for l in prompt.split('\n') if l.strip()]
        for line in reversed(lines):
            if '=' not in line and len(line) >= 5:
                query = line
                break
        else:
            return None, None
    else:
        query = query_match.group(1).strip()
    
    # Remove spaces from query
    query = query.replace(" ", "")
    
    if len(query) != 5:
        return None, None
    
    lines: List[str] = []
    
    # Header
    lines.append("We need to infer the transformation rule from the examples.")
    lines.append("I will analyze the structure and detect the operation pattern.")
    lines.append("I will put my final answer inside \\boxed{}.")
    lines.append("")
    
    # Step 1: Parse all examples
    lines.append("Step 1: Parse all examples")
    lines.append("")
    
    examples: List[_Example] = []
    
    for lhs, rhs in example_lines:
        lhs = lhs.strip().replace(" ", "")
        rhs = rhs.strip().replace(" ", "")
        
        if len(lhs) != 5:
            continue
        
        ex = _Example(
            left=(lhs[0], lhs[1]),
            op=lhs[2],
            right=(lhs[3], lhs[4]),
            output=rhs
        )
        examples.append(ex)
        
        lines.append(f"{_quote(lhs)} = {_quote(rhs)}")
        lines.append(f"  Input: {_box(lhs)}")
        lines.append(f"  Left operand: {_quote(ex.left[0])}{_quote(ex.left[1])}")
        lines.append(f"  Operator: {_quote(ex.op)}")
        lines.append(f"  Right operand: {_quote(ex.right[0])}{_quote(ex.right[1])}")
        lines.append(f"  Output: {_box(rhs)}")
        lines.append("")
    
    if not examples:
        return None, None
    
    # Step 2: Group examples by operator
    lines.append("Step 2: Group examples by operator")
    lines.append("")
    
    by_operator: dict[str, List[_Example]] = {}
    for ex in examples:
        by_operator.setdefault(ex.op, []).append(ex)
    
    for op, op_examples in by_operator.items():
        lines.append(f"Operator {_quote(op)}: {len(op_examples)} example(s)")
    lines.append("")
    
    # Step 3: Test concatenation patterns for each example
    lines.append("Step 3: Test concatenation patterns")
    lines.append("")
    
    concat_types: dict[str, str] = {}
    
    for ex in examples:
        left_str = ex.left[0] + ex.left[1]
        right_str = ex.right[0] + ex.right[1]
        
        fwd = left_str + right_str
        rev = right_str + left_str
        
        is_fwd = (ex.output == fwd)
        is_rev = (ex.output == rev)
        
        lines.append(f"Example: {_quote(left_str + ex.op + right_str)} = {_quote(ex.output)}")
        lines.append(f"  Forward concatenation: {_box(fwd)} {'match' if is_fwd else 'mismatch'}")
        lines.append(f"  Reverse concatenation: {_box(rev)} {'match' if is_rev else 'mismatch'}")
        lines.append("")
    
    # Step 4: Determine operator rules
    lines.append("Step 4: Determine operator rules")
    lines.append("")
    
    for op, op_examples in by_operator.items():
        ct = _concat_type(op_examples)
        
        if ct == "fwd":
            concat_types[op] = "fwd"
            lines.append(f"Operator {_quote(op)}: forward concatenation (AB{op}CD = ABCD)")
        elif ct == "rev":
            concat_types[op] = "rev"
            lines.append(f"Operator {_quote(op)}: reverse concatenation (AB{op}CD = CDAB)")
        else:
            lines.append(f"Operator {_quote(op)}: pattern unclear, defaulting to forward")
            concat_types[op] = "fwd"
    
    lines.append("")
    
    # Step 5: Parse and solve query
    lines.append("Step 5: Apply rule to query")
    lines.append("")
    
    query_left = (query[0], query[1])
    query_op = query[2]
    query_right = (query[3], query[4])
    
    lines.append(f"Query: {_quote(query)}")
    lines.append(f"  Input: {_box(query)}")
    lines.append(f"  Left operand: {_quote(query_left[0])}{_quote(query_left[1])}")
    lines.append(f"  Operator: {_quote(query_op)}")
    lines.append(f"  Right operand: {_quote(query_right[0])}{_quote(query_right[1])}")
    lines.append("")
    
    # Determine operation type for query
    if query_op in concat_types:
        q_type = concat_types[query_op]
        lines.append(f"The operator {_quote(query_op)} is known: {q_type} concatenation")
    else:
        q_type = "fwd"
        lines.append(f"The operator {_quote(query_op)} is unknown, defaulting to forward concatenation")
    
    lines.append("")
    
    # Compute result
    left_str = query_left[0] + query_left[1]
    right_str = query_right[0] + query_right[1]
    
    if q_type == "fwd":
        computed = left_str + right_str
        lines.append(f"  Forward concatenation({_quote(left_str)}, {_quote(right_str)})")
    else:
        computed = right_str + left_str
        lines.append(f"  Reverse concatenation({_quote(left_str)}, {_quote(right_str)})")
    
    lines.append(f"  = {_box(computed)}")
    lines.append(f"  Output: {_quote(computed)}")
    lines.append("")
    # Final answer
    final_answer = computed
    
    lines.append("Final Answer:")
    lines.append(f"The answer is \\boxed{{{final_answer}}}")
    
    trace = "\n".join(lines)
    
    return final_answer, trace

import re
import math
from typing import Optional, Tuple, List
from collections import defaultdict
from dataclasses import dataclass


@dataclass
class FoundOperation:
    """Detected operation for an operator."""
    name: str
    reverse_operands: bool
    reverse_result: bool
    format_type: str  # 'num', 'neg_suffix', 'neg_prefix', 'pre'
    operator_char: str


def _safe_eval(fn, a: int, b: int) -> Optional[int]:
    """Safely evaluate a function, returning None on error."""
    try:
        result = fn(a, b)
        return int(result) if result is not None else None
    except:
        return None


def _reverse_str(s: str) -> str:
    """Reverse a string, preserving negative sign."""
    if s.startswith("-"):
        return "-" + s[1:][::-1]
    return s[::-1]


def _common_operations(a: int, b: int) -> List[Tuple[str, int]]:
    """Return list of (name, result) for common operations."""
    ops = []
    ops.append(("addition", a + b))
    ops.append(("subtraction (a-b)", a - b))
    ops.append(("reverse subtraction (b-a)", b - a))
    ops.append(("multiplication", a * b))
    ops.append(("absolute difference", abs(a - b)))
    ops.append(("negated absolute difference", -abs(a - b)))
    ops.append(("concatenation", int(str(abs(a)) + str(abs(b)))))
    ops.append(("reverse concatenation", int(str(abs(b)) + str(abs(a)))))
    return ops


def _rare_operations(a: int, b: int) -> List[Tuple[str, int]]:
    """Return list of (name, result) for rare operations."""
    ops = []
    ops.append(("multiply+1", a * b + 1))
    ops.append(("multiply-1", a * b - 1))
    ops.append(("add+1", a + b + 1))
    ops.append(("add-1", a + b - 1))
    
    if b != 0:
        ops.append(("division (a/b)", a // b))
        ops.append(("modulo (a mod b)", a % b))
    
    if a != 0 and b != 0:
        ops.append(("gcd", math.gcd(abs(a), abs(b))))
    
    # Two-digit specific operations
    sa, sb = str(abs(a)), str(abs(b))
    if len(sa) == 2 and len(sb) == 2:
        d1, d2, d3, d4 = int(sa[0]), int(sa[1]), int(sb[0]), int(sb[1])
        ops.append(("digit sum diff", (d1 + d2) - (d3 + d4)))
        ops.append(("cross multiply", d1 * d3 + d2 * d4))
        ops.append(("determinant", d1 * d4 - d2 * d3))
    
    return ops


def _all_operations(a: int, b: int) -> List[Tuple[str, int]]:
    """Return all operations (common first, then rare)."""
    return _common_operations(a, b) + _rare_operations(a, b)


def _format_expr(op_name: str, a: str, b: str) -> str:
    """Format the mathematical expression for display."""
    if op_name == "addition":
        return f"{a} + {b}"
    elif op_name == "subtraction (a-b)":
        return f"{a} - {b}"
    elif op_name == "multiplication":
        return f"{a} * {b}"
    elif op_name == "absolute difference":
        return f"|{a} - {b}|"
    elif op_name == "concatenation":
        return f"{a} || {b}"
    elif "determinant" in op_name and len(a) == 2 and len(b) == 2:
        return f"{a[0]}*{b[1]} - {a[1]}*{b[0]}"
    return f"{op_name}({a}, {b})"

        lines_out.append(f"  Operation: {op_rule.name}")
        lines_out.append(f"  Expression: {expr}")
        lines_out.append(f"  Computed: {computed}")
        if op_rule.format_type != "num":
            lines_out.append(f"  Formatted: {final_answer}")
        lines_out.append("")
    
    # Final answer
    
    lines_out.append("Final Answer:")
    lines_out.append(f"The answer is \\boxed{{{final_answer}}}")
    
    
    trace = "\n".join(lines_out)
    
    return final_answer, trace

import re
from typing import Optional, Tuple, List


def solve_bit_manipulation_with_trace(prompt: str) -> Tuple[Optional[str], Optional[str]]:
    """
    Generate COT trace for bit manipulation with simplified but effective reasoning.
    
    Focuses on the most common patterns:
    - Identity (I)
    - NOT
    - Simple binary operations with stride-1 patterns
    """
    # Extract examples
    examples = re.findall(r"(\d{8})\s*->\s*(\d{8})", prompt)
    if not examples:
        return None, None
    
    # Extract question
    q = re.search(r"(?:transform|convert|apply to)\s*(\d{8})", prompt, re.IGNORECASE)
    if not q:
        # Try to find last 8-bit number as question
        all_8bit = re.findall(r"\b(\d{8})\b", prompt)
        if not all_8bit or len(all_8bit) < len(examples) + 1:
            return None, None
        question = all_8bit[-1]
    else:
        question = q.group(1)
    
    if len(question) != 8 or not all(c in '01' for c in question):
        return None, None
    
    inputs = [inp for inp, _ in examples]
    outputs = [out for _, out in examples]
    
    lines: List[str] = []
    
    # Header
    lines.append("We need to deduce the bit transformation rule from examples.")
    lines.append("I will analyze each output bit position to find its rule.")
    lines.append("I will put my final answer inside \\boxed{}.")
    lines.append("")
    
    # Step 1: Show examples
    lines.append("Step 1: Examine the examples")
    for i, (inp, out) in enumerate(examples):
        lines.append(f"Example {i+1}: {inp} → {out}")
    lines.append("")
    
    # Step 2: Analyze bit columns
    lines.append("Step 2: Analyze each output bit position")
    lines.append("")
    
    rules: List[str] = []
    rule_exprs: List[str] = []
    
    for bit_pos in range(8):
        lines.append(f"Output bit {bit_pos}:")
        
        # Get output column
        out_col = ''.join(outputs[j][bit_pos] for j in range(len(outputs)))
        lines.append(f"  Output pattern: {out_col}")
        
        # Try identity (I)
        found_rule = None
        for src in range(8):
            in_col = ''.join(inputs[j][src] for j in range(len(inputs)))
            if in_col == out_col:
                found_rule = f"I{src}"
                lines.append(f"  Input bit {src}: {in_col} ← Match!")
                lines.append(f"  Rule: bit{bit_pos}_out = bit{src}_in")
                break
        
        # Try NOT
        if not found_rule:
            for src in range(8):
                in_col = ''.join(inputs[j][src] for j in range(len(inputs)))
                in_col_inv = ''.join('1' if c == '0' else '0' for c in in_col)
                if in_col_inv == out_col:
                    found_rule = f"NOT{src}"
                    lines.append(f"  Input bit {src}: {in_col}")
                    lines.append(f"  Inverted: {in_col_inv} ← Match!")
                    lines.append(f"  Rule: bit{bit_pos}_out = NOT(bit{src}_in)")
                    break
        
        # Try XOR with adjacent bits
        if not found_rule:
            for a in range(8):
                for b in range(a+1, min(a+4, 8)):  # Only check nearby bits
                    xor_col = ''
                    for j in range(len(inputs)):
                        val_a = inputs[j][a]
                        val_b = inputs[j][b]
                        xor_col += '1' if val_a != val_b else '0'
                    if xor_col == out_col:
                        found_rule = f"XOR{a}{b}"
                        lines.append(f"  XOR(bit{a}, bit{b}): {xor_col} ← Match!")
                        lines.append(f"  Rule: bit{bit_pos}_out = XOR(bit{a}_in, bit{b}_in)")
                        break
                if found_rule:
                    break
        
        # Default: constant
        if not found_rule:
            if out_col == '0' * len(outputs):
                found_rule = "C0"
                lines.append(f"  Rule: bit{bit_pos}_out = 0 (constant)")
            elif out_col == '1' * len(outputs):
                found_rule = "C1"
                lines.append(f"  Rule: bit{bit_pos}_out = 1 (constant)")
            else:
                found_rule = "DEFAULT"
                lines.append(f"  Rule: bit{bit_pos}_out = 1 (default)")
        
        rules.append(found_rule)
        rule_exprs.append(found_rule)
        lines.append("")
    
    # Step 3: Detect patterns
    lines.append("Step 3: Detect stride patterns")
    
    # Check for stride-1 identity pattern
    identity_stride = True
    for i in range(len(rules) - 1):
        if rules[i].startswith('I') and rules[i+1].startswith('I'):
            src1 = int(rules[i][1:])
            src2 = int(rules[i+1][1:])
            if (src2 - src1) % 8 != 1:
                identity_stride = False
                break
        else:
            identity_stride = False
            break
    
    if identity_stride and rules[0].startswith('I'):
        offset = int(rules[0][1:])
        lines.append(f"  Identity with stride +1, offset {offset}")
        lines.append(f"  Pattern: bit_out[i] = bit_in[(i + {offset}) % 8]")
    else:
        lines.append("  No simple stride pattern detected")
        lines.append("  Using per-bit rules")
    lines.append("")
    
    # Step 4: Apply to question
    lines.append(f"Step 4: Apply transformation to {question}")
    lines.append("")
    
    lines.append("Input bits:")
    for i, bit in enumerate(question):
        lines.append(f"  bit{i} = {bit}")
    lines.append("")
    
    lines.append("Applying rules:")
    result_bits = []
    for i, rule in enumerate(rules):
        if rule.startswith('I'):
            src = int(rule[1:])
            val = question[src]
            lines.append(f"  bit{i}_out = bit{src}_in = {val}")
            result_bits.append(val)
        elif rule.startswith('NOT'):
            src = int(rule[3:])
            val = question[src]
            inv_val = '1' if val == '0' else '0'
            lines.append(f"  bit{i}_out = NOT(bit{src}_in) = NOT({val}) = {inv_val}")
            result_bits.append(inv_val)
        elif rule.startswith('XOR'):
            a = int(rule[3])
            b = int(rule[4])
            val_a = question[a]
            val_b = question[b]
            xor_val = '1' if val_a != val_b else '0'
            lines.append(f"  bit{i}_out = XOR(bit{a}_in, bit{b}_in) = XOR({val_a}, {val_b}) = {xor_val}")
            result_bits.append(xor_val)
        elif rule == 'C0':
            lines.append(f"  bit{i}_out = 0")
            result_bits.append('0')
        elif rule == 'C1':
            lines.append(f"  bit{i}_out = 1")
            result_bits.append('1')
        else:  # DEFAULT
            lines.append(f"  bit{i}_out = 1 (default)")
            result_bits.append('1')
    
    lines.append("")
    computed_answer = ''.join(result_bits)
    lines.append(f"Output bits: {computed_answer}")
    lines.append("")
    
    final_answer = computed_answer
    
    lines.append("Final Answer:")
    lines.append(f"The answer is \\boxed{{{final_answer}}}")
    
    trace = "\n".join(lines)
    return final_answer, trace

def is_answer_correct(computed_ans: str, ground_truth: str, rel_tol: float = 1e-2) -> bool:
    """
    模拟官方评测逻辑，验证生成的答案是否正确。
    """
    if computed_ans is None or ground_truth is None:
        return False
        
    # 去除首尾空格进行清理
    c_ans = str(computed_ans).strip()
    gt_ans = str(ground_truth).strip()

    # 1. 优先进行严格的字符串匹配 (处理文本、十六进制等)
    if c_ans == gt_ans:
        return True

    # 2. 如果字符串不匹配，尝试转化为浮点数进行容差比较 (处理物理、单位换算等)
    try:
        c_float = float(c_ans)
        gt_float = float(gt_ans)
        
        # 使用 math.isclose 进行相对误差计算 (rel_tol)
        # 比赛说明中被隐藏的数字大概率是 1e-3, 1e-4 或 1e-5，你可以按需调整
        return math.isclose(c_float, gt_float, rel_tol=rel_tol)
    except ValueError:
        # 如果无法转换为浮点数（说明是纯文本类题目且没匹配上），直接判错
        return False

import pandas as pd

def process_and_track_data(df, dataset_name="Train"):
    """
    处理数据并生成统计报告的封装函数
    """
    records = []
    
    # 初始化统计指标
    stats = {
        "total": 0,
        "kept": 0,
        "dropped_logic": 0,  # 因为返回 None 被踢出的
        "dropped_error": 0,  # 因为代码崩溃抛出 Exception 被踢出的
        "category_kept": Counter(),
        "category_dropped": Counter(),
        "failed_details": [] # 记录被踢出的详细信息，方便你 debug 脚本
    }

    print(f"\n🚀 开始处理 {dataset_name} 集...")
    
    for idx, row in df.iterrows():
        stats["total"] += 1
        prompt = row["prompt"]
        gt_answer = str(row["answer"])
        sid = str(row["id"])
        
        try:
            category, final_answer, trace = build_thinking_trace(prompt, gt_answer, sample_id=sid)
            
            # 防御性编程：如果脚本崩溃返回了 None，尝试从原始 row 中获取类别
            cat_label = category if category else row.get("category", "Unknown")
            
            # 规则拦截：COT程序无法正确解答
            if final_answer is None or trace is None:
                stats["dropped_logic"] += 1
                stats["category_dropped"][cat_label] += 1
                stats["failed_details"].append({"id": sid, "category": cat_label, "reason": "Logic/Trace returned None"})
                continue
            
            # 成功保留
            records.append(format_train_row(prompt, trace))
            stats["kept"] += 1
            stats["category_kept"][cat_label] += 1
            
        except Exception as e:
            stats["dropped_error"] += 1
            # 尝试获取类别，获取不到记为 Exception_Unknown
            cat_label = row.get("category", "Exception_Unknown") 
            stats["category_dropped"][cat_label] += 1
            stats["failed_details"].append({"id": sid, "category": cat_label, "reason": f"Exception: {str(e)}"})
            # 建议将这里的 print 注释掉，否则遇到大面积报错会刷屏。我们统一在最后看报告。
            # print(f"Error processing sample {sid}: {e}")

    # === 打印华丽的统计报告 ===
    print("=" * 50)
    print(f"📊 {dataset_name} 集清洗统计报告")
    print("=" * 50)
    print(f"总样本数: {stats['total']}")
    
    if stats['total'] > 0:
        kept_ratio = stats['kept'] / stats['total']
        print(f"✅ 成功保留: {stats['kept']} ({kept_ratio:.1%})")
        print(f"❌ 逻辑踢出: {stats['dropped_logic']}")
        print(f"⚠️ 异常报错: {stats['dropped_error']}")
        print("-" * 50)
        print("📈 各类别保留情况 (保留数 / 总数):")
        
        # 合并所有出现过的类别
        all_cats = set(stats["category_kept"].keys()).union(set(stats["category_dropped"].keys()))
        
        for cat in sorted(all_cats):
            k = stats["category_kept"][cat]
            d = stats["category_dropped"][cat]
            t = k + d
            cat_ratio = k / t if t > 0 else 0
            
            # 根据保留率给个小提示，低于 50% 标红报警（这里用文字提示）
            alert = " 🚨 (需优化脚本!)" if cat_ratio < 0.5 else ""
            print(f"  - {cat:<20}: {k:<4} / {t:<4} ({cat_ratio:>6.1%}){alert}")
    print("=" * 50 + "\n")

    return Dataset.from_list(records), stats
