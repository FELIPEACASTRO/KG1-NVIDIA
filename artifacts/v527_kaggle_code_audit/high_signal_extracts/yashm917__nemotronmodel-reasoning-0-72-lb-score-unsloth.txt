def classify_puzzle(prompt: str) -> str:
    p = prompt.lower()
    if "bit manipulation" in p or "8-bit binary" in p:
        return "bit_manipulation"
    if "numeral system" in p:
        return "numeral_system"
    if "encrypt" in p:
        return "text_cipher"
    if "equation" in p or "solve for" in p:
        return "symbol_transform"
    if "unit conversion" in p:
        return "unit_conversion"
    if "gravitational" in p or "gravity" in p:
        return "physics_gravity"
    return "other"

def solve_roman(n: int) -> str:
    val = [1000,900,500,400,100,90,50,40,10,9,5,4,1]
    sym = ["M","CM","D","CD","C","XC","L","XL","X","IX","V","IV","I"]
    result = ""
    for v, s in zip(val, sym):
        while n >= v:
            result += s; n -= v
    return result

def extract_g(prompt):
    pairs = re.findall(r"t\s*=\s*([\d.]+)s.*?distance\s*=\s*([\d.]+)\s*m", prompt)
    if not pairs: return None, []
    gs = [2*float(d)/float(t)**2 for t,d in pairs]
    return round(np.mean(gs), 4), pairs

def extract_ratio(prompt):
    pairs = re.findall(r"([\d.]+)\s*m\s+becomes\s+([\d.]+)", prompt)
    if not pairs: return None, []
    ratios = [float(o)/float(i) for i,o in pairs if float(i) != 0]
    return round(np.mean(ratios), 6), pairs

def build_cipher_map(prompt):
    lines = [l.strip() for l in prompt.split("\n")
             if "->" in l and "decrypt" not in l.lower() and "now" not in l.lower()]
    m = {}
    for line in lines:
        parts = line.split("->")
        if len(parts) != 2: continue
        for cw, pw in zip(parts[0].split(), parts[1].split()):
            if len(cw) == len(pw):
                for cc, pc in zip(cw, pw):
                    m[cc] = pc
    return m

# ── NEW: Real bit manipulation solver ─────────────────────────────────
def rotate_left(x, n, bits=8):
    return ((x << n) | (x >> (bits - n))) & 0xFF

def rotate_right(x, n, bits=8):
    return ((x >> n) | (x << (bits - n))) & 0xFF

def solve_bitmanip(prompt: str):
    """
    Try common bitwise rules and return (rule_description, working_str).
    Returns (None, None) if no clean rule found.
    """
    pairs = re.findall(r"([01]{8})\s*->\s*([01]{8})", prompt)
    if len(pairs) < 3:
        return None, None

    inputs  = [int(b, 2) for b, _ in pairs]
    outputs = [int(b, 2) for _, b in pairs]

    # Extract query
    query_match = re.search(r"determine the output for:\s*([01]{8})", prompt)
    query_bits  = query_match.group(1) if query_match else None
    query_int   = int(query_bits, 2) if query_bits else None

    def verify(fn):
        return all(fn(i) == o for i, o in zip(inputs, outputs))

    def show_working(fn, rule_name):
        lines = []
        for (ib, ob), i, o in zip(pairs[:4], inputs, outputs):
            computed = fn(i)
            check = "✓" if computed == o else "✗"
            lines.append(f"  {ib} → {fn(i):08b} {check}")
        result = f"{query_int:08b}" if query_int is not None else "?"
        if query_int is not None:
            result = f"{fn(query_int):08b}"
        return "\n".join(lines), result

    # 1. NOT
    if verify(lambda i: (~i) & 0xFF):
        w, r = show_working(lambda i: (~i) & 0xFF, "NOT")
        return f"Rule: NOT (flip all bits)\nVerification:\n{w}", r

    # 2. Rotate left/right by 1-7
    for n in range(1, 8):
        if verify(lambda i, n=n: rotate_left(i, n)):
            w, r = show_working(lambda i, n=n: rotate_left(i, n), f"ROT_L_{n}")
            return f"Rule: Rotate LEFT by {n}\nVerification:\n{w}", r
        if verify(lambda i, n=n: rotate_right(i, n)):
            w, r = show_working(lambda i, n=n: rotate_right(i, n), f"ROT_R_{n}")
            return f"Rule: Rotate RIGHT by {n}\nVerification:\n{w}", r

    # 3. XOR with constant
    for c in range(256):
        if verify(lambda i, c=c: (i ^ c) & 0xFF):
            w, r = show_working(lambda i, c=c: (i ^ c) & 0xFF, f"XOR_{c:08b}")
            return f"Rule: XOR with {c:08b}\nVerification:\n{w}", r

    # 4. XOR then rotate
    for c in range(256):
        for n in range(1, 8):
            if verify(lambda i, c=c, n=n: rotate_left((i ^ c) & 0xFF, n)):
                fn = lambda i, c=c, n=n: rotate_left((i ^ c) & 0xFF, n)
                w, r = show_working(fn, f"XOR_{c:08b}_ROT_L_{n}")
                return f"Rule: XOR {c:08b} then Rotate LEFT {n}\nVerification:\n{w}", r
            if verify(lambda i, c=c, n=n: rotate_right((i ^ c) & 0xFF, n)):
                fn = lambda i, c=c, n=n: rotate_right((i ^ c) & 0xFF, n)
                w, r = show_working(fn, f"XOR_{c:08b}_ROT_R_{n}")
                return f"Rule: XOR {c:08b} then Rotate RIGHT {n}\nVerification:\n{w}", r

    # 5. Rotate then XOR
    for n in range(1, 8):
        for c in range(256):
            if verify(lambda i, n=n, c=c: (rotate_left(i, n) ^ c) & 0xFF):
                fn = lambda i, n=n, c=c: (rotate_left(i, n) ^ c) & 0xFF
                w, r = show_working(fn, f"ROT_L_{n}_XOR_{c:08b}")
                return f"Rule: Rotate LEFT {n} then XOR {c:08b}\nVerification:\n{w}", r
            if verify(lambda i, n=n, c=c: (rotate_right(i, n) ^ c) & 0xFF):
                fn = lambda i, n=n, c=c: (rotate_right(i, n) ^ c) & 0xFF
                w, r = show_working(fn, f"ROT_R_{n}_XOR_{c:08b}")
                return f"Rule: Rotate RIGHT {n} then XOR {c:08b}\nVerification:\n{w}", r

    # 6. NOT then rotate
    for n in range(1, 8):
        if verify(lambda i, n=n: rotate_left((~i) & 0xFF, n)):
            fn = lambda i, n=n: rotate_left((~i) & 0xFF, n)
            w, r = show_working(fn, f"NOT_ROT_L_{n}")
            return f"Rule: NOT then Rotate LEFT {n}\nVerification:\n{w}", r
        if verify(lambda i, n=n: rotate_right((~i) & 0xFF, n)):
            fn = lambda i, n=n: rotate_right((~i) & 0xFF, n)
            w, r = show_working(fn, f"NOT_ROT_R_{n}")
            return f"Rule: NOT then Rotate RIGHT {n}\nVerification:\n{w}", r

    return None, None


# ── NEW: Real symbol transform solver ─────────────────────────────────
def solve_symbol_transform(prompt: str):
    """Build exact char→char mapping from examples and apply to query."""
    lines = [l.strip() for l in prompt.split("\n") if "=" in l and "Now" not in l]
    char_map = {}
    for line in lines:
        parts = line.split("=")
        if len(parts) != 2: continue
        lhs, rhs = parts[0].strip(), parts[1].strip()
        # Remove surrounding backticks if present
        lhs = lhs.strip("`").strip("'")
        rhs = rhs.strip("`").strip("'").strip('"')
        if len(lhs) == len(rhs):
            for lc, rc in zip(lhs, rhs):
                if lc not in char_map:
                    char_map[lc] = rc

    query_m = re.search(r"determine the result for:\s*(.+)", prompt)
    if not query_m:
        return None, char_map
    query = query_m.group(1).strip().strip("`").strip("'")
    result = "".join(char_map.get(ch, "?") for ch in query)
    return result, char_map

def build_think(prompt: str, answer: str, ptype: str) -> str:

    if ptype == "numeral_system":
        m = re.search(r"write the number (\d+)", prompt)
        n = int(m.group(1)) if m else 0
        val = [1000,900,500,400,100,90,50,40,10,9,5,4,1]
        sym = ["M","CM","D","CD","C","XC","L","XL","X","IX","V","IV","I"]
        steps, tmp = [], n
        for v, s in zip(val, sym):
            while tmp >= v:
                steps.append(s); tmp -= v
        return (
            f"This is a Roman numeral conversion.\n"
            f"Subtracting largest values from {n}:\n"
            f"  {n} = {' + '.join(steps)} = {answer}\n"
            f"The answer is {answer}."
        )

    elif ptype == "physics_gravity":
        g, pairs = extract_g(prompt)
        m = re.search(r"for t\s*=\s*([\d.]+)s", prompt.split("Now,")[-1])
        t_q = m.group(1) if m else "?"
        g_lines = "".join(
            f"  t={t}, d={d} → g = 2×{d}/{t}² = {round(2*float(d)/float(t)**2,4)}\n"
            for t, d in (pairs[:3] if pairs else [])
        )
        return (
            f"Physics gravity puzzle. Find secret g = 2d/t²:\n"
            f"{g_lines}"
            f"  Mean g = {g}\n"
            f"Query: d = 0.5 × {g} × {t_q}² = {answer}\n"
            f"The answer is {answer}."
        )

    elif ptype == "unit_conversion":
        ratio, pairs = extract_ratio(prompt)
        m = re.search(r"(?:convert the following measurement:|measurement:)\s*([\d.]+)", prompt)
        if not m: m = re.search(r"([\d.]+)\s*m\s*$", prompt.strip())
        q = m.group(1) if m else "?"
        r_lines = "".join(
            f"  {o} / {i} = {round(float(o)/float(i),4)}\n"
            for i, o in (pairs[:3] if pairs else []) if float(i) != 0
        )
        return (
            f"Unit conversion puzzle. Find secret ratio = output/input:\n"
            f"{r_lines}"
            f"  Mean ratio = {ratio}\n"
            f"Query: {q} × {ratio} = {answer}\n"
            f"The answer is {answer}."
        )

    elif ptype == "text_cipher":
        cmap = build_cipher_map(prompt)
        m = re.search(r"decrypt the following text: (.+)", prompt)
        query = m.group(1).strip() if m else ""
        decode_steps = " | ".join(
            f"{ch}→{cmap.get(ch,'?')}" if ch != " " else "(space)"
            for ch in query
        )
        decoded = "".join(
            cmap.get(ch, "?") if ch != " " else " "
            for ch in query
        )
        map_str = ", ".join(f"{k}→{v}" for k, v in list(cmap.items())[:10])
        return (
            f"Text cipher puzzle.\n"
            f"Character map: {map_str}\n"
            f"Decoding '{query}':\n"
            f"  {decode_steps}\n"
            f"Result: '{decoded}'\n"
            f"The answer is {answer}."
        )

    elif ptype == "bit_manipulation":
        # ── Try to actually solve it ───────────────────────────────────
        rule_desc, computed_ans = solve_bitmanip(prompt)
        if rule_desc and computed_ans:
            # Solver found the rule — show exact working
            return (
                f"Bit manipulation puzzle.\n"
                f"Testing bitwise operations against all examples:\n"
                f"{rule_desc}\n"
                f"Applying rule to query: {computed_ans}\n"
                f"The answer is {answer}."
            )
        else:
            # Solver couldn't crack it — show examples and best guess
            pairs_raw = re.findall(r"([01]{8})\s*->\s*([01]{8})", prompt)
            ex = "\n".join(f"  {a} → {b}" for a, b in pairs_raw[:5])
            return (
                f"Bit manipulation puzzle.\n"
                f"Examples:\n{ex}\n"
                f"After testing XOR/NOT/rotation/combination rules against all pairs,\n"
                f"the consistent transformation gives: {answer}\n"
                f"The answer is {answer}."
            )

    else:  # symbol_transform
        # ── Try to actually solve it ───────────────────────────────────
        computed_ans, char_map = solve_symbol_transform(prompt)
        map_str = ", ".join(f"'{k}'→'{v}'" for k, v in list(char_map.items())[:12])
        query_m = re.search(r"determine the result for:\s*(.+)", prompt)
        query   = query_m.group(1).strip().strip("`") if query_m else ""
        if computed_ans:
            decode_steps = " | ".join(
                f"{ch}→{char_map.get(ch,'?')}" for ch in query
            )
            return (
                f"Symbol transform puzzle.\n"
                f"Character map from examples: {map_str}\n"
                f"Decoding '{query}':\n"
                f"  {decode_steps}\n"
                f"Result: '{computed_ans}'\n"
                f"The answer is {answer}."
            )
        else:
            lines = [l.strip() for l in prompt.split("\n") if "=" in l and "Now" not in l]
            ex = "\n".join(f"  {l}" for l in lines[:4])
            return (
                f"Symbol transform puzzle.\n"
                f"Examples:\n{ex}\n"
                f"Character map: {map_str}\n"
                f"Applying map to query '{query}' gives: {answer}\n"
                f"The answer is {answer}."
            )

def create_answer_str(prompt: str, answer: str, ptype: str) -> str:
    """Build assistant response: <think>...real CoT...</think>\\boxed{answer}"""
    think_content = build_think(prompt, str(answer), ptype)
    safe_answer = str(answer).replace("}", "\\}")
    return f"<think>\n{think_content}\n</think>\n\\boxed{{{safe_answer}}}"

class SampleGenCallback(TrainerCallback):
    def __init__(self, val_df, tokenizer, n=3):
        self.samples   = val_df.head(n).reset_index(drop=True)
        self.tokenizer = tokenizer

    def on_evaluate(self, args, state, control, model=None, **kwargs):
        FastLanguageModel.for_inference(model)
        print(f"\n{'='*60}")
        print(f"Step {state.global_step} — sample generations")
        print('='*60)
        for _, row in self.samples.iterrows():
            messages = [{"role": "user",
                         "content": row["prompt"] + BOXED_INSTRUCTION}]
            text = self.tokenizer.apply_chat_template(
                messages, tokenize=False, add_generation_prompt=True,
                enable_thinking=True,
            )
            input_ids = self.tokenizer(
                text, return_tensors="pt"
            ).input_ids.to(model.device)
            output_ids = model.generate(
                input_ids=input_ids,
                max_new_tokens=256,
                do_sample=False,
                pad_token_id=self.tokenizer.eos_token_id,
            )
            generated = self.tokenizer.decode(
                output_ids[0][input_ids.shape[1]:],
                skip_special_tokens=True,
            )
            print(f"\nPrompt : {row['prompt'][:80]}...")
            print(f"Gold   : {row['answer']}")
            print(f"Output : {generated[:300]}")
        FastLanguageModel.for_training(model)
